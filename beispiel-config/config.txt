# DMX Bridge – Grundeinstellungen
# Gespeicherte Änderungen übernimmt das Plugin automatisch (nach ca. 1 s).
# Alles hinter # ist Kommentar.
format = 1

# Art-Net-Ziel: IP des Nodes (Unicast empfohlen, z. B. 192.168.0.50).
# 255.255.255.255 = Broadcast an alle im Netz.
artnet_ip = 255.255.255.255
artnet_port = 6454

# Wiederholrate pro Universum in Hz (1–44).
# Änderungen werden zusätzlich sofort gesendet (max. ~44x pro Sekunde).
refresh_hz = 40

# MIDI-CCs als Werte annehmen: CC-Nummer = Slot (wie die Note), Wert 0–127 → 0–255.
# Praktisch für Fades über CC-Lanes im MIDI-Editor.
cc_eingang = an

# MIDI-Kanal → Universum + Block (wie die Input-Blocks am X32)
#   Block 1 = DMX   1–128   Block 2 = DMX 129–256
#   Block 3 = DMX 257–384   Block 4 = DMX 385–512
#   Note 0 = erster Kanal des Blocks, Note 127 = letzter.
#   "aus" = MIDI-Kanal wird ignoriert.
#   Universum = Art-Net-Port-Address (0 = erstes Universum).
midi1  = 0 1
midi2  = 0 2
midi3  = 0 3
midi4  = 0 4
midi5  = 1 1
midi6  = 1 2
midi7  = 1 3
midi8  = 1 4
midi9  = aus
midi10 = aus
midi11 = aus
midi12 = aus
midi13 = aus
midi14 = aus
midi15 = aus
midi16 = aus
